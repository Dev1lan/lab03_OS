cmake_minimum_required(VERSION 3.16)
project(parent C)
find_package(Threads REQUIRED)
add_executable(parent src/sum_arrays.c)
target_compile_features(parent PRIVATE c_std_11)
target_link_libraries(parent PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(parent PRIVATE /W4)
else()
    target_compile_options(parent PRIVATE -Wall -Wextra -Wpedantic)
endif()
